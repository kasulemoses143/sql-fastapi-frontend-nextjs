## Learning objectives

Create relationships using IDs; return joined labels; protect referenced records; run regression tests.

## Before you begin

Commit working code before replacing a complete file. A recovery checkpoint creates code in a new folder; it does not copy credentials or initialize database data.

## Vocabulary

Foreign-key violation: referenced parent missing. Regression test: detects broken existing behavior. Fixture: isolated test setup. Dependency override: substitute test resources.

## Explanation

An enrollment's two IDs come from existing records. POST checks that each parent exists and then inserts the pair. The unique pair remains necessary because two requests can pass a precheck at the same time. A joined GET response contains names for display without duplicating them in the enrollment table.

Deleting a student/course with enrollments returns 409. Change a status with PATCH; remove the enrollment before deleting its parent. These are explicit product rules, not incidental database errors.

Tests use isolated resources. The local fallback uses SQLite with foreign keys enabled for quick API regression checks; PostgreSQL verification is still required because database engines differ. The CI workflow uses actual PostgreSQL and applies the handwritten SQL first.

## Worked procedure

1. Replace main.py with the complete final backend/main.py below. Read list_enrollments and connect each join to Lesson 07.
2. Create an enrollment using real IDs. Try a duplicate pair (409), missing parents (404), invalid status (422) and deleting a referenced student (409).
3. Change status to completed; verify the joined list. Delete the enrollment, then delete the student and course.
4. Copy the complete tests folder from the reference using the files below. Run `python -m pytest -q` from backend.
5. Read TESTING.md and run the PostgreSQL path against a disposable test database. Never target the development or production database with reset-enabled tests.



### Complete file backend/main.py

Create or replace `backend/main.py` inside **student-project**. This is the entire file for this stage.

```python
import os
from typing import Annotated
from fastapi import Depends, FastAPI, HTTPException, Query, Response
from fastapi.middleware.cors import CORSMiddleware
from sqlalchemy import select
from sqlalchemy.exc import IntegrityError
from sqlalchemy.orm import Session
from db import get_session
from models import Student, Course, Enrollment
from schemas import (
    StudentWrite, StudentRead, CourseWrite, CourseRead,
    EnrollmentWrite, EnrollmentUpdate, EnrollmentRead, EnrollmentDetail,
)

app = FastAPI(title='Student Course Enrollment API', version='1.0.0')
origins = [x.strip() for x in os.getenv('CORS_ORIGINS', 'http://localhost:3000').split(',') if x.strip()]
app.add_middleware(
    CORSMiddleware, allow_origins=origins, allow_credentials=False,
    allow_methods=['GET', 'POST', 'PUT', 'PATCH', 'DELETE'],
    allow_headers=['Content-Type'],
)
SessionDep = Annotated[Session, Depends(get_session)]
Limit = Annotated[int, Query(ge=1, le=100)]
Offset = Annotated[int, Query(ge=0)]

def find_or_404(session, model, record_id):
    row = session.get(model, record_id)
    if row is None:
        raise HTTPException(404, 'Record not found')
    return row

def commit_or_conflict(session):
    try:
        session.commit()
    except IntegrityError:
        session.rollback()
        raise HTTPException(409, 'Conflict with an existing record or relationship')

@app.get('/health')
def health():
    return {'status': 'ok'}

@app.get('/students', response_model=list[StudentRead])
def list_students(session: SessionDep, limit: Limit = 100, offset: Offset = 0):
    return session.scalars(select(Student).order_by(Student.id).limit(limit).offset(offset)).all()

@app.post('/students', response_model=StudentRead, status_code=201)
def create_student(body: StudentWrite, session: SessionDep):
    row = Student(**body.model_dump())
    session.add(row)
    commit_or_conflict(session)
    session.refresh(row)
    return row

@app.get('/students/{record_id}', response_model=StudentRead)
def get_student(record_id: int, session: SessionDep):
    return find_or_404(session, Student, record_id)

@app.put('/students/{record_id}', response_model=StudentRead)
def update_student(record_id: int, body: StudentWrite, session: SessionDep):
    row = find_or_404(session, Student, record_id)
    row.full_name = body.full_name
    row.email = str(body.email)
    commit_or_conflict(session)
    session.refresh(row)
    return row

@app.delete('/students/{record_id}', status_code=204)
def delete_student(record_id: int, session: SessionDep):
    row = find_or_404(session, Student, record_id)
    session.delete(row)
    commit_or_conflict(session)
    return Response(status_code=204)

@app.get('/courses', response_model=list[CourseRead])
def list_courses(session: SessionDep, limit: Limit = 100, offset: Offset = 0):
    return session.scalars(select(Course).order_by(Course.id).limit(limit).offset(offset)).all()

@app.post('/courses', response_model=CourseRead, status_code=201)
def create_course(body: CourseWrite, session: SessionDep):
    row = Course(**body.model_dump())
    session.add(row)
    commit_or_conflict(session)
    session.refresh(row)
    return row

@app.get('/courses/{record_id}', response_model=CourseRead)
def get_course(record_id: int, session: SessionDep):
    return find_or_404(session, Course, record_id)

@app.put('/courses/{record_id}', response_model=CourseRead)
def update_course(record_id: int, body: CourseWrite, session: SessionDep):
    row = find_or_404(session, Course, record_id)
    row.code, row.title, row.credits = body.code, body.title, body.credits
    commit_or_conflict(session)
    session.refresh(row)
    return row

@app.delete('/courses/{record_id}', status_code=204)
def delete_course(record_id: int, session: SessionDep):
    row = find_or_404(session, Course, record_id)
    session.delete(row)
    commit_or_conflict(session)
    return Response(status_code=204)

@app.get('/enrollments', response_model=list[EnrollmentDetail])
def list_enrollments(session: SessionDep, limit: Limit = 100, offset: Offset = 0):
    query = (
        select(Enrollment, Student.full_name, Course.code, Course.title)
        .join(Student, Student.id == Enrollment.student_id)
        .join(Course, Course.id == Enrollment.course_id)
        .order_by(Enrollment.id).limit(limit).offset(offset)
    )
    return [dict(
        id=e.id, student_id=e.student_id, course_id=e.course_id,
        enrolled_on=e.enrolled_on, status=e.status,
        student_name=name, course_code=code, course_title=title,
    ) for e, name, code, title in session.execute(query).all()]

@app.post('/enrollments', response_model=EnrollmentRead, status_code=201)
def create_enrollment(body: EnrollmentWrite, session: SessionDep):
    find_or_404(session, Student, body.student_id)
    find_or_404(session, Course, body.course_id)
    row = Enrollment(**body.model_dump())
    session.add(row)
    commit_or_conflict(session)
    session.refresh(row)
    return row

@app.get('/enrollments/{record_id}', response_model=EnrollmentRead)
def get_enrollment(record_id: int, session: SessionDep):
    return find_or_404(session, Enrollment, record_id)

@app.patch('/enrollments/{record_id}', response_model=EnrollmentRead)
def update_enrollment(record_id: int, body: EnrollmentUpdate, session: SessionDep):
    row = find_or_404(session, Enrollment, record_id)
    row.status = body.status
    commit_or_conflict(session)
    session.refresh(row)
    return row

@app.delete('/enrollments/{record_id}', status_code=204)
def delete_enrollment(record_id: int, session: SessionDep):
    row = find_or_404(session, Enrollment, record_id)
    session.delete(row)
    commit_or_conflict(session)
    return Response(status_code=204)
```

### Complete file backend/tests/conftest.py

Create or replace `backend/tests/conftest.py` inside **student-project**. This is the entire file for this stage.

```python
import os
import sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).resolve().parents[1]))
# The application engine is never used by tests: dependency overrides supply a test session.
os.environ.setdefault('DATABASE_URL', 'postgresql+psycopg://unused:unused@localhost/unused')
os.environ['CORS_ORIGINS'] = 'http://localhost:3000'
import pytest
from fastapi.testclient import TestClient
from sqlalchemy import create_engine, event
from sqlalchemy.orm import Session
from sqlalchemy.pool import StaticPool
from db import Base, get_session, database_url
from main import app

@pytest.fixture
def client():
    raw = os.getenv('TEST_DATABASE_URL')
    if raw:
        if os.getenv('ALLOW_TEST_DATABASE_RESET') != 'yes':
            raise RuntimeError('Use a disposable database and set ALLOW_TEST_DATABASE_RESET=yes')
        engine = create_engine(database_url(raw))
    else:
        engine = create_engine('sqlite://', connect_args={'check_same_thread':False}, poolclass=StaticPool)
        @event.listens_for(engine, 'connect')
        def foreign_keys(connection, _):
            connection.execute('PRAGMA foreign_keys=ON')
    # The reset flag is explicit because this removes the course tables.
    Base.metadata.drop_all(engine)
    if raw:
        schema = Path(__file__).resolve().parents[1] / 'sql/001_schema.sql'
        with engine.connect() as connection:
            connection.exec_driver_sql(schema.read_text())
            connection.commit()
    else:
        Base.metadata.create_all(engine)
    def override():
        with Session(engine) as session:
            yield session
    app.dependency_overrides[get_session] = override
    try:
        with TestClient(app) as test_client:
            yield test_client
    finally:
        app.dependency_overrides.clear()
        Base.metadata.drop_all(engine)
        engine.dispose()
```

```python
def student(client, name='Amina', email='amina@example.com'):
    response = client.post('/students', json={'full_name':name, 'email':email})
    assert response.status_code == 201
    return response.json()

def course(client):
    response = client.post('/courses', json={'code':'sql101', 'title':'SQL', 'credits':3})
    assert response.status_code == 201
    return response.json()

def test_health_and_student_crud(client):
    assert client.get('/health').json() == {'status':'ok'}
    s = student(client)
    assert client.get(f"/students/{s['id']}").json()['full_name'] == 'Amina'
    updated = client.put(f"/students/{s['id']}", json={'full_name':'Amina Demo','email':'AMINA@example.com'})
    assert updated.status_code == 200
    assert updated.json()['email'] == 'amina@example.com'
    assert len(client.get('/students').json()) == 1
    assert client.delete(f"/students/{s['id']}").status_code == 204
    assert client.get(f"/students/{s['id']}").status_code == 404

def test_validation_and_duplicate_recovery(client):
    student(client)
    assert client.post('/students', json={'full_name':'Other','email':'AMINA@example.com'}).status_code == 409
    assert student(client, 'Brian','brian@example.com')['id'] > 0
    for body in [dict(full_name=' ',email='ok@example.com'),dict(full_name='X',email='bad'),dict(full_name='X',email='x@example.com',id=1)]:
        assert client.post('/students',json=body).status_code == 422
    assert client.get('/students?limit=101').status_code == 422
    assert client.get('/students?offset=-1').status_code == 422

def test_course_crud(client):
    c = course(client)
    assert c['code'] == 'SQL101'
    assert client.get(f"/courses/{c['id']}").status_code == 200
    assert client.post('/courses',json={'code':'sql101','title':'Other','credits':3}).status_code == 409
    assert client.put(f"/courses/{c['id']}",json={'code':'API101','title':'API','credits':4}).status_code == 200
    assert client.post('/courses',json={'code':'BAD','title':'Bad','credits':7}).status_code == 422
    assert client.post('/courses',json={'code':'BAD','title':'Bad','credits':3.5}).status_code == 422
    assert client.delete(f"/courses/{c['id']}").status_code == 204
    assert client.get(f"/courses/{c['id']}").status_code == 404

def test_enrollment_integrity_and_join(client):
    s,c = student(client),course(client)
    body = {'student_id':s['id'], 'course_id':c['id']}
    r = client.post('/enrollments',json=body)
    assert r.status_code == 201
    e = r.json()
    assert e['enrolled_on'] and e['status'] == 'active'
    assert client.get(f"/enrollments/{e['id']}").status_code == 200
    assert client.post('/enrollments',json=body).status_code == 409
    assert client.delete(f"/students/{s['id']}").status_code == 409
    assert client.delete(f"/courses/{c['id']}").status_code == 409
    assert client.get('/enrollments').json()[0]['student_name'] == 'Amina'
    assert client.patch(f"/enrollments/{e['id']}",json={'status':'completed'}).status_code == 200
    assert client.patch(f"/enrollments/{e['id']}",json={'status':'wrong'}).status_code == 422
    assert client.delete(f"/enrollments/{e['id']}").status_code == 204
    assert client.delete(f"/students/{s['id']}").status_code == 204
    assert client.delete(f"/courses/{c['id']}").status_code == 204

def test_missing_records_and_pagination(client):
    assert client.get('/students/99999').status_code == 404
    assert client.post('/enrollments',json={'student_id':99999,'course_id':99999}).status_code == 404
    student(client)
    student(client,'Brian','brian@example.com')
    rows = client.get('/students?limit=1&offset=1').json()
    assert len(rows) == 1 and rows[0]['full_name'] == 'Brian'

def test_cors(client):
    headers={'Origin':'http://localhost:3000','Access-Control-Request-Method':'POST','Access-Control-Request-Headers':'content-type'}
    r = client.options('/students',headers=headers)
    assert r.status_code == 200
    assert r.headers['access-control-allow-origin'] == 'http://localhost:3000'
    headers['Origin'] = 'https://untrusted.example'
    r = client.options('/students',headers=headers)
    assert 'access-control-allow-origin' not in r.headers
```
